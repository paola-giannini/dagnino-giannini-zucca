########## Rust versione ricorsiva 

type Link = Option<Box<Node>>;

struct Node {
    value: i32,
    next: Link,
}

pub struct LinkedList {
    head: Link,
}

impl LinkedList {
    pub fn new() -> Self {
        LinkedList { head: None }
    }

    pub fn append(&mut self, val: i32) {
        Self::recursive_append(&mut self.head, val);
    }

    fn recursive_append(link: &mut Link, val: i32) {
        match link {
            // Base case: we found the end of the list
            None => {
                *link = Some(Box::new(Node {
                    value: val,
                    next: None,
                }));
            }
            // Recursive step: move to the next node
            Some(node) => {
                Self::recursive_append(&mut node.next, val);
            }

fn insert_rec(link: &mut Option<Box<Node<T>>>, index: usize, value: T) {
        if index == 0 {
            let new_node = Box::new(Node {
                value,
                next: link.take(), // 
            });
            *link = Some(new_node);
        } else {
            match link {
                Some(node) => Self::insert_rec(&mut node.next, index - 1, value),
                None => panic!("Index out of bounds"),
            }
        }
    }
