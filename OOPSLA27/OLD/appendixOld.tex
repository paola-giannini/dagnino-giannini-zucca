\section{ Proofs of Soundness Theorems}

{\em The sum of types} is defined by:
\begin{quote}
$\Graded{\tau}{\Rgr_1}{\rgr_1}\rsum\Graded{\tau}{\Rgr_2}{\rgr_2}=\Graded{\tau}{(\Rgr_1 \rsub \rgr_2) \rmeet (\Rgr_2 \rsub \rgr_1)}{(\rgr_1\rsum\rgr_2)}$ 
\end{quote}
Note that sum of types is undefined if they have different base type ($\tau$) or if the needed subtractions of grades are undefined. Sum is associative and commutative. 

\begin{proposition}\label[proposition]{prop:sum}\ 
\begin{enumerate}
\item  \label{prop:sum:assoc}  $(\T_1\rsum\T_2)\rsum\T_3=\T_1\rsum(\T_2\rsum\T_3)$
\item \label{prop:sum:comm}  $\T_1\rsum\T_2=\T_2\rsum\T_1$
\item \label{prop:sum:zero}  If $\T=\Graded{\tau}{\Rgr}{\rgr}$, then $\T_{\rzero}=\Graded{\tau}{\Rgr\rsum\rgr}{\rzero}$ is such that $\T\rsum\T_{\rzero}=\T_{\rzero}\rsum\T=\T$. 
\end{enumerate}
\end{proposition} 
\begin{proof}\hfill
\begin{enumerate}
\item
Let $\Graded{\tau}{\Tgr}{\tgr}=(\Graded{\tau}{\Rgr_1}{\rgr_1}\rsum\Graded{\tau}{\Rgr_2}{\rgr_2})\rsum\Graded{\tau}{\Rgr_3}{\rgr_3}$ and
$\Graded{\tau}{\Sgr}{\sgr}=\Graded{\tau}{\Rgr_1}{\rgr_1}\rsum(\Graded{\tau}{\Rgr_2}{\rgr_2}\rsum\Graded{\tau}{\Rgr_3}{\rgr_3})$. By associativity of $\rsum$,
\begin{center}
$\tgr=(\rgr_1\rsum\rgr_2)\rsum\rgr_3=\rgr_1\rsum(\rgr_2\rsum\rgr_3)=\sgr$.
\end{center}
Consider now the assumption grades:
\begin{center}
$
\begin{array}{lcl}
\Tgr&=& (((\Rgr_1 \rsub \rgr_2) \rmeet (\Rgr_2 \rsub \rgr_1))\rsub\rgr_3)\rmeet(\Rgr_3 \rsub (\rgr_1\rsum\rgr_2))\\
&=& (((\Rgr_1 \rsub \rgr_2)\rsub\rgr_3) \rmeet ((\Rgr_2 \rsub \rgr_1)\rsub\rgr_3))\rmeet(\Rgr_3 \rsub (\rgr_1\rsum\rgr_2))\text{ by \refItem{prop:unique-alg}{3}}\\
&=& (((\Rgr_1 \rsub \rgr_2)\rsub\rgr_3) \rmeet ((\Rgr_2 \rsub \rgr_1)\rsub\rgr_3))\rmeet((\Rgr_3 \rsub\rgr_2)\rsub\rgr_1)\text{ by \refItem{prop:subtractive-alg}{4}}\\
&=&((\Rgr_1 \rsub( \rgr_2\rsum\rgr_3)) \rmeet ((\Rgr_2 \rsub \rgr_1)\rsub\rgr_3))\rmeet((\Rgr_3 \rsub\rgr_2)\rsub\rgr_1)\text{ by \refItem{prop:subtractive-alg}{4}}\\
&=& ((\Rgr_1 \rsub( \rgr_2\rsum\rgr_3)) \rmeet ((\Rgr_2 \rsub \rgr_3)\rsub\rgr_1))\rmeet((\Rgr_3 \rsub\rgr_2)\rsub\rgr_1)\text{ by \refItem{prop:subtractive-alg}{5}}\\
&=& (\Rgr_1 \rsub( \rgr_2\rsum\rgr_3)) \rmeet (((\Rgr_2 \rsub \rgr_3)\rsub\rgr_1)\rmeet((\Rgr_3 \rsub\rgr_2)\rsub\rgr_1))\text{ by associativity of $\rmeet$}\\
&=& (\Rgr_1 \rsub( \rgr_2\rsum\rgr_3)) \rmeet (((\Rgr_2 \rsub \rgr_3)\rmeet(\Rgr_3 \rsub\rgr_2)) \rsub\rgr_1)\text{ by \refItem{prop:unique-alg}{3}}\\
&=&\Sgr
\end{array}
$
\end{center}
\item Immediate from commutativity of $\rsum$ and $\rmeet$.
\item From \refItem{prop:subtractive-alg}{1} $\Rgr\rsub\rzero=\Rgr$ and $\Rgr\rord(\Rgr\rsum\rgr)\rsub\rgr$ from \refItem{prop:subtractive-alg}{3}. Therefore
$(\Rgr\rsub\rzero)\rmeet((\Rgr\rsum\rgr)\rsub\rgr)=\Rgr$ and $\T\rsum\T_{\rzero}=\T$. From \cref{prop:sum:comm}  we get $\T_{\rzero}\rsum\T=\T$.0
\end{enumerate}
\end{proof}
Associativity and commutativity of sum between contexts of variables and references follows from \cref{prop:sum}.

\begin{lemma}\label[lemma]{lem:typeSumCtx}\ 
\begin{enumerate}
\item \label{lem:typeSumCtx:comp}
Let $\CtxElem{\x}{\Rgr}{\rgr}{\tau}\in\Gamma$ and $\Gamma_1\ctxsum\Gamma_2\ctxord\Gamma$. If $\CtxElem{\x}{\Sgr_i}{\sgr_i}{\tau_i}\in\Gamma_i$, then $\tau_i=\tau$ and $\Rgr\ctxord(\Sgr_1\rsub\sgr_2)\rmeet(\Sgr_2\rsub\sgr_1)$  and $\sgr_1\rsum\sgr_2\rord\rgr$,
i.e. $\Graded{\tau}{\Rgr}{\rgr}\subt \Graded{\tau}{\Sgr_1}{\sgr_1}\rsum\Graded{\tau}{\Sgr_2}{\sgr_2}$.
\item \label{lem:typeSumCtx:decomp} Let $\T=\Graded{\tau}{\Rgr}{\rgr}$ and $\rgr_1$ and $\rgr_2$ be such that $\rgr_1\rsum\rgr_2\rord\rgr$,
$\T'=\Graded{\tau}{\Rgr\rsum\sgr}{\sgr}=\Graded{\tau}{\Rgr\rsum\rgr_2}{\rgr_1}\rsum\Graded{\tau}{\Rgr\rsum\rgr_1}{\rgr_2}$ is defined and $\T\subt\T'$.
\end{enumerate}
\end{lemma}
\begin{proof}
\begin{enumerate}
\item By definition of sum and preorder of contexts. 
\item From \refItem{prop:subtractive-alg}{3},  $\Rgr\rord(\Rgr\rsum\rgr_2)\rsub\rgr_2$ and $\Rgr\rord(\Rgr\rsum\rgr_1)\rsub\rgr_1$.
Let $\Sgr=((\Rgr\rsum\rgr_2)\rsub\rgr_2)\rmeet((\Rgr\rsum\rgr_1)\rsub\rgr_1)$, we have that 
$\T'= \Graded{\tau}{\Sgr}{\rgr_1\rsum\rgr_2}$. From $\Rgr\rord\Sgr$ and $\rgr_1\rsum\rgr_2\rord\rgr$ we derive $\T\subt\T'$.
\end{enumerate}
\end{proof}

\begin{lemma}\label{lem:lessSigma}
If $\OK{\alpha_\Sigma}{\upsilon_\Sigma}{\mem}$ and $\Sigma'\ctxord\Sigma$, then  $\OK{\alpha_{\Sigma'}}{\upsilon_{\Sigma'}}{\mem}$.
\end{lemma}
\begin{proof}  
Without loss of generality we can assume that $\dom{\Sigma}=\dom{\Sigma'}=\dom{\mem}$.\\
From $\Sigma'\ctxord\Sigma$ we get that: if $\VarGrade{\loc}{\Rgr}\in\alpha_{\Sigma'}$  and $\VarGrade{\loc}{\rgr}\in\upsilon_{\Sigma'}$, then 
$\VarGrade{\loc}{\Sgr}\in\alpha_{\Sigma}$  and $\VarGrade{\loc}{\sgr}\in\upsilon_{\Sigma}$ and $\rgr\rord\sgr$ and $\Sgr\rord\Rgr$.
Therefore for all $\loc, \locPrime\in\dom{\Sigma}$ if $\GrReach{\loc}{\alpha_{\Sigma'}}{}{\Rgr}{}{\locPrime}$, then 
$\GrReach{\loc}{\alpha_{\Sigma}}{}{\Sgr}{}{\locPrime}$ with $\Sgr\rord\Rgr$. Moroever, if $\locPrime_i\in\dom{\Sigma}$, for $i\in 1..n$,
and $\GrReach{\locPrime_1,\ldots,\locPrime_n}{}{\upsilon_{\Sigma'}}{}{\sgr}{\iota}$,
then $\GrReach{\locPrime_1,\ldots,\locPrime_n}{}{\upsilon_{\Sigma}}{}{\rgr}{\iota}$ with $\rgr\rord\sgr$. Therefore, for all $\loc\in\dom{\mem}$
if the predicate holds for $\Pair{\alpha_{\Sigma}}{\upsilon_{\Sigma}}$ is also holds for $\Pair{\alpha_{\Sigma'}}{\upsilon_{\Sigma'}}$
\end{proof}
